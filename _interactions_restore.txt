function initPageModularLayoutEditor(){
  const site=document.getElementById('promoterSite');
  if(!site||site.dataset.modularLayoutBound==='1')return;
  site.dataset.modularLayoutBound='1';

  site.addEventListener('change',e=>{
    const sel=e.target.closest?.('.page-module-width');
    if(!sel||!pageDetailsEditMode||experienceMode!=='edit')return;
    e.preventDefault();e.stopPropagation();
    setPageModuleWidth(sel.closest('.site-content-card'),sel.value)
  },true);

  site.addEventListener('click',e=>{
    if(!pageDetailsEditMode||experienceMode!=='edit')return;

    const own=e.target.closest?.('.page-module-own-row');
    if(own){
      e.preventDefault();e.stopPropagation();e.stopImmediatePropagation?.();
      togglePageModuleOwnRow(own.closest('.site-content-card'));
      return
    }

    const image=e.target.closest?.('.page-module-image');
    if(image){
      e.preventDefault();e.stopPropagation();e.stopImmediatePropagation?.();
      choosePageModuleImage(image.closest('.site-content-card'));
      return
    }

    const removeImage=e.target.closest?.('.page-module-remove-image');
    if(removeImage){
      e.preventDefault();e.stopPropagation();e.stopImmediatePropagation?.();
      removePageModuleImage(removeImage.closest('.site-content-card'));
      return
    }

    const line=e.target.closest?.('.page-module-line');
    if(line){
      e.preventDefault();e.stopPropagation();e.stopImmediatePropagation?.();
      togglePageModuleSeparator(line.closest('.site-content-card'))
    }
  },true);

  site.addEventListener('mousedown',e=>{
    const handle=e.target.closest?.('.page-module-move');
    if(!handle||!pageDetailsEditMode||experienceMode!=='edit'||e.button!==0)return;
    const card=handle.closest('.site-content-card');if(!card)return;

    initialisePageModuleGroup(card);
    handle.draggable=true;
    handle.setAttribute('draggable','true');
    card.draggable=false;

    pageModuleDragElement=card;
    pageModuleDragKey=pageModuleKey(card)
  },true);

  site.addEventListener('pointerdown',e=>{
    const handle=e.target.closest?.('.page-module-move');
    if(!handle||!pageDetailsEditMode||experienceMode!=='edit')return;
    const card=handle.closest('.site-content-card');if(!card)return;

    initialisePageModuleGroup(card);
    handle.draggable=true;
    handle.setAttribute('draggable','true');
    card.draggable=false;

    pageModuleDragElement=card;
    pageModuleDragKey=pageModuleKey(card)
  },true);

  site.addEventListener('dragstart',e=>{
    const handle=e.target.closest?.('.page-module-move');
    const card=handle?.closest?.('.site-content-card') || e.target.closest?.('.site-content-card');

    if(!handle||!card||card!==pageModuleDragElement){
      e.preventDefault();
      return
    }

    card.classList.add('page-module-dragging');
    document.body.classList.add('page-module-native-drag-active');

    e.dataTransfer.effectAllowed='move';
    try{e.dataTransfer.setData('text/plain',pageModuleDragKey)}catch(_){}
  },true);

  site.addEventListener('dragover',e=>{
    if(!pageModuleDragKey||!pageDetailsEditMode)return;
    const target=e.target.closest?.('.site-content-card');
    const dragged=pageModuleDragElement;
    if(!target||!dragged||target===dragged)return;

    // This applies to every page tab and each matching Overview mirror,
    // but never lets one page's DOM content leak into another page.
    if(pageModuleScopeForCard(target)!==pageModuleScopeForCard(dragged))return;

    e.preventDefault();
    clearPageModuleDropClasses(site);

    const side=pageModuleDropSide(target,e);
    target.classList.add(
      side==='before'?'page-module-drop-before':'page-module-drop-after',
      side==='before'?'page-module-pair-left':'page-module-pair-right',
      'page-module-cross-drop'
    )
  },true);

  site.addEventListener('drop',e=>{
    if(!pageModuleDragKey||!pageDetailsEditMode)return;
    const target=e.target.closest?.('.site-content-card');
    const dragged=pageModuleDragElement;
    if(!target||!dragged||target===dragged)return;
    if(pageModuleScopeForCard(target)!==pageModuleScopeForCard(dragged))return;

    e.preventDefault();e.stopPropagation();

    const sourceParent=dragged.parentElement;
    const targetParent=target.parentElement;
    const side=pageModuleDropSide(target,e);

    initialisePageModuleGroup(dragged);
    initialisePageModuleGroup(target);

    // The actual DOM placement defines the pair order.
    targetParent.insertBefore(dragged,side==='before'?target:target.nextSibling);

    const draggedRec=ensurePageModuleRecord(dragged,{enable:true});
    if(draggedRec)draggedRec.group=pageModuleGroupKey(targetParent);

    // Critical V152 behaviour: a side drop means "put these beside each other".
    // Disable OWN ROW for both, and make them 50/50 if their existing widths
    // would otherwise force the second block onto the next row.
    pairPageModules(dragged,target);
    targetParent.classList.add('page-modular-grid');

    savePageModuleOrder(targetParent);
    if(sourceParent&&sourceParent!==targetParent)savePageModuleOrder(sourceParent);

    clearPageModuleDropClasses(site);
    applyPageModularLayout?.();
    applyPageSpacingToDom?.();
    refreshPageEmptyFlow?.();
    persistPageModularLayout();

    // Rebuild Overview immediately so the same layout change is visible there.
    if(pageModuleScopeForCard(dragged)?.dataset?.siteSection!=='explore'){
      renderOverviewRollup?.();
      applyPageModularLayout?.();
      applyPageSpacingToDom?.()
    }
  },true);

  site.addEventListener('dragend',e=>{
    const handle=e.target.closest?.('.page-module-move');
    const card=handle?.closest?.('.site-content-card') || e.target.closest?.('.site-content-card');

    if(card)card.classList.remove('page-module-dragging');
    if(handle){
      handle.draggable=true;
      handle.setAttribute('draggable','true')
    }

    document.body.classList.remove('page-module-native-drag-active');
    pageModuleDragKey='';
    pageModuleDragElement=null;
    clearPageModuleDropClasses(site)
  },true);

  applyPageModularLayout()
}

let pageHiddenSections={};

function capturePageHiddenSectionState(){
  return JSON.parse(JSON.stringify(pageHiddenSections||{}))
}
function applyPageHiddenSectionState(saved){
  pageHiddenSections=saved&&typeof saved==='object'?JSON.parse(JSON.stringify(saved)):{};
  applyPageDetailSectionVisibility?.()
}
function loadLocalPageHiddenSectionState(){
  try{
    const raw=localStorage.getItem('mothershipPageHiddenSectionsV1');
    applyPageHiddenSectionState(raw?JSON.parse(raw):{})
  }catch(e){applyPageHiddenSectionState({})}
}
function pageDetailSectionKey(card){return pageModuleKey(card)}
function applyPageDetailSectionVisibility(root=document){
  if(!root?.querySelectorAll)return;
  const cards=[];
  if(root.matches?.('.site-content-card'))cards.push(root);
  root.querySelectorAll?.('.site-content-card').forEach(card=>cards.push(card));

  cards.forEach(card=>{
    const key=pageDetailSectionKey(card);if(!key)return;
    card.dataset.pageDetailSectionKey=key;
    const hidden=pageHiddenSections[key]===true;
    card.classList.toggle('page-detail-section-hidden',hidden);

    let btn=[...card.children].find(el=>el.classList?.contains('page-detail-section-control'))||null;
    if(!btn){
      btn=document.createElement('button');
      btn.type='button';
      btn.className='page-detail-section-control';
      card.appendChild(btn)
    }
    btn.dataset.sectionKey=key;
    btn.textContent=hidden?'RESTORE SECTION':'DELETE SECTION';
    btn.title=hidden?'Restore this section':'Hide this entire section from Performance Mode'
  });
  applyPageModularLayout?.(root);
  refreshPageEmptyFlow?.(root);
  syncRestoreHiddenControl?.()
}

function currentPageHiddenKeys(){
  const section=currentPageSpacingSection?.()||currentPerformancePageSection?.()||'venue';
  const prefix=section+'::';
  return Object.keys(pageHiddenSections||{}).filter(k=>k.startsWith(prefix)&&pageHiddenSections[k]===true)
}
function syncRestoreHiddenControl(){
  const btn=document.getElementById('pageRestoreHidden');if(!btn)return;
  const count=currentPageHiddenKeys().length;
  btn.textContent=count?'RESTORE HIDDEN · '+count:'RESTORE HIDDEN';
  btn.classList.toggle('has-hidden',count>0);
  btn.disabled=count===0
}
function restoreHiddenSectionsForCurrentPage(){
  currentPageHiddenKeys().forEach(key=>delete pageHiddenSections[key]);
  persistPageHiddenSections?.();
  applyPageDetailSectionVisibility?.();
  renderOverviewRollup?.();
  applyPageDetailSectionVisibility?.();
  applyPageSpacingToDom?.();
  applyPageFrameToDom?.();
  refreshPageEmptyFlow?.();
  syncRestoreHiddenControl()
}

function persistPageHiddenSections(){
  try{
    localStorage.setItem('mothershipPageHiddenSectionsV1',JSON.stringify(capturePageHiddenSectionState()))
  }catch(e){}
  if(typeof setProjectDirty==='function'&&projectReady)setProjectDirty(true)
}
function togglePageDetailSection(card){
  const key=pageDetailSectionKey(card);if(!key)return;
  if(pageHiddenSections[key]===true)delete pageHiddenSections[key];
  else pageHiddenSections[key]=true;
  persistPageHiddenSections();
  applyPageDetailSectionVisibility?.();
  renderOverviewRollup?.();
  applyPageDetailSectionVisibility?.()
}
function initPageDetailSectionEditor(){
  const site=document.getElementById('promoterSite');
  if(!site||site.dataset.sectionDeleteBound==='1')return;
  site.dataset.sectionDeleteBound='1';

  site.addEventListener('click',e=>{
    const btn=e.target.closest?.('.page-detail-section-control');
    if(!btn)return;
    if(!pageDetailsEditMode||experienceMode!=='edit')return;
    e.preventDefault();e.stopPropagation();e.stopImmediatePropagation?.();
    const card=btn.closest('.site-content-card');if(card)togglePageDetailSection(card)
  },true);

  applyPageDetailSectionVisibility()
}

let pageDetailsEditMode=false;

function syncPageDetailsEditControls(){
  const btn=document.getElementById('perfPageInlineEdit');
  if(btn){
    btn.textContent=pageDetailsEditMode?'PAGE DETAILS EDITING…':'SHOW PAGE DETAILS TO EDIT';
    btn.classList.toggle('active',pageDetailsEditMode)
  }
}
function savePageTextEdits(){
  if(inlinePageTextActiveEl)finishInlinePageTextEdit();

  try{localStorage.setItem('mothershipPageInlineEditsV1',JSON.stringify(capturePageInlineEditState()))}catch(e){}
  try{localStorage.setItem('mothershipPageInlineStylesV1',JSON.stringify(capturePageInlineStyleState()))}catch(e){}
  try{localStorage.setItem('mothershipPageHiddenSectionsV1',JSON.stringify(capturePageHiddenSectionState()))}catch(e){}
  try{localStorage.setItem('mothershipPageModularLayoutV1',JSON.stringify(capturePageModularLayoutState()))}catch(e){}
  try{localStorage.setItem('mothershipPageSpacingV1',JSON.stringify(capturePageSpacingState()))}catch(e){}
  try{localStorage.setItem('mothershipPageFrameV1',JSON.stringify(capturePageFrameState()))}catch(e){}
  try{localStorage.setItem('mothershipPerformancePagesV1',JSON.stringify(capturePerformancePageState()))}catch(e){}

  /*
    Individual tab pages are authoritative. Rebuild them first, then regenerate
    Overview from those live page sources so Overview can never drift out of sync.
  */
  Object.keys(PERFORMANCE_PAGE_SECTION_IDS)
    .filter(section=>section!=='explore')
    .forEach(section=>renderPerformancePage(section,{skipRollup:true}));

  renderPerformancePage('explore',{skipRollup:true});
  renderOverviewRollup?.();

  applyPageInlineEditsToDom?.();
  applyPageInlineStylesToDom?.();
  applyPageDetailSectionVisibility?.();
  applyPageModularLayout?.();
  applyPageSpacingToDom?.();
  refreshPageEmptyFlow?.();
  dedupePageContentModules?.();
  dedupePageEditorArtifacts?.();

  saveLocalEditState?.(false);
  syncRestoreHiddenControl?.();
  syncPageGapControl?.();

  [document.getElementById('perfPageSaveText'),document.getElementById('pageDetailsSaveText')]
    .filter(Boolean).forEach(btn=>{
      if(!btn.dataset.normalLabel)btn.dataset.normalLabel=btn.textContent;
      btn.textContent='SAVED ✓';
      clearTimeout(btn._pageSaveTimer);
      btn._pageSaveTimer=setTimeout(()=>btn.textContent=btn.dataset.normalLabel||'SAVE PAGE DETAILS',1100)
    })
}
function setPageDetailsEditMode(on,{restoreEditor=true}={}){
  on=!!on;
  if(on&&document.body.classList.contains('promoter-link'))return;
  if(on===pageDetailsEditMode){syncPageDetailsEditControls();return}

  if(on){
    if(experienceMode!=='edit')setExperienceMode('edit');
    pageDetailsEditMode=true;
    setEditMode(false);
    document.body.classList.add('page-details-edit-mode','performance-mode');
    setSitePreviewState(true);

    const section=currentPerformancePageSection?.()||'venue';
    siteSetActiveNav(section);
    siteShowSection(section);
    siteSetSectionCover(section);
    if(section==='explore')document.body.classList.remove('site-info-expanded','site-info-peek');
    else document.body.classList.add('site-info-expanded');
    if(section!=='explore')ensureSiteInfoTitlePanel?.(section);
    applyPageInlineEditsToDom?.();
    applyPageInlineStylesToDom?.();
    applyPageDetailSectionVisibility?.();
    applyPageModularLayout?.();
    applyPageSpacingToDom?.();
    refreshPageEmptyFlow?.();
    dedupePageEditorArtifacts?.();
    syncPageGapControl?.();
    syncPageWidthControl?.();
    applyPageFrameToDom?.();
    syncRestoreHiddenControl?.();

    const detailsDock=document.getElementById('pageDetailsEditDock');
    if(detailsDock)detailsDock.dataset.activePage=siteActiveSection||'overview';

    setTimeout(()=>{
      window.scrollTo({top:section==='explore'?0:siteInfoTargetY(section),left:0,behavior:'auto'});
      updateSiteBackToVenueChip?.()
    },30)
  }else{
    if(inlinePageTextActiveEl)finishInlinePageTextEdit();
    savePageTextEdits();
    pageDetailsEditMode=false;
    document.body.classList.remove('page-details-edit-mode','inline-page-text-active');
    setSitePreviewState(false);
    document.body.classList.remove('performance-mode');
    if(experienceMode==='edit'&&restoreEditor)setEditMode(true);
    window.scrollTo({top:0,left:0,behavior:'auto'})
  }
  syncPageDetailsEditControls()
